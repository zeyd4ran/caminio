🧠 3. LLM Wrapper (Shared Brain)

core/llm.py

from openai import OpenAI

client = OpenAI(api_key="YOUR_API_KEY")

def ask_llm(prompt):
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[{"role": "user", "content": prompt}]
    )
    return response.choices[0].message.content

🔁 9. Main Agent Loop (THE HEART)

agent_loop.py

from agents.planner import planner
from agents.coder import coder
from agents.debugger import debugger
from core.executor import run_code
from core.memory import load_memory, save_memory

goal = "Build a task management system"

while True:
    memory = load_memory()

    print("Thinking...")

    task = planner(goal, memory)
    print("TASK:", task)

    code = coder(task)

    with open("app.py", "w") as f:
        f.write(code)

    returncode, stdout, stderr = run_code()

    if returncode != 0:
        print("Error detected")
        fixed_code = debugger(stderr, code)

        with open("app.py", "w") as f:
            f.write(fixed_code)

        continue

    memory["completed"].append(task)
    save_memory(memory)

    if "complete" in task.lower():
        break
